cmake_minimum_required(VERSION 3.25)
project(oc_protobuf LANGUAGES C CXX)

option(OPENCHANGER_DEV "" OFF)
include(${CMAKE_CURRENT_LIST_DIR}/../flags.cmake)

set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_CONFORMANCE OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_PROTOC_BINARIES ON CACHE BOOL "" FORCE)
set(protobuf_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(protobuf_DISABLE_RTTI ON CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(protobuf_INSTALL ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)

add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../thirdparty/protobuf protobuf)
foreach(target libprotobuf libprotobuf-lite libprotoc protoc)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 17)
    target_compile_options(${target} PRIVATE /GR- /EHs-c-)
    target_compile_definitions(${target} PRIVATE _HAS_EXCEPTIONS=0)
    get_target_property(sources ${target} SOURCES)
    list(FILTER sources EXCLUDE REGEX "version[.]rc$")
    set_property(TARGET ${target} PROPERTY SOURCES ${sources})
endforeach()
